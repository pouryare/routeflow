-- trigger order_lines_void_figures_delete on order_lines
CREATE TRIGGER order_lines_void_figures_delete AFTER DELETE ON order_lines BEGIN UPDATE orders SET kg_total = NULL, total_toman = NULL, cost_toman = NULL WHERE id = OLD.order_id; END;

-- trigger order_lines_void_figures_insert on order_lines
CREATE TRIGGER order_lines_void_figures_insert AFTER INSERT ON order_lines BEGIN UPDATE orders SET kg_total = NULL, total_toman = NULL, cost_toman = NULL WHERE id = NEW.order_id; END;

-- trigger order_lines_void_figures_update on order_lines
CREATE TRIGGER order_lines_void_figures_update AFTER UPDATE ON order_lines BEGIN UPDATE orders SET kg_total = NULL, total_toman = NULL, cost_toman = NULL WHERE id IN (OLD.order_id, NEW.order_id); END;

-- trigger user_roles_after_insert on users
CREATE TRIGGER user_roles_after_insert AFTER INSERT ON users BEGIN INSERT OR IGNORE INTO user_roles (user_id, role) VALUES (NEW.id, NEW.role); END;

-- trigger user_roles_after_update on users
CREATE TRIGGER user_roles_after_update AFTER UPDATE OF role ON users WHEN (SELECT COUNT(*) FROM user_roles WHERE user_id = NEW.id) <= 1 AND EXISTS (SELECT 1 FROM user_roles WHERE user_id = NEW.id AND role = OLD.role) BEGIN UPDATE user_roles SET role = NEW.role WHERE user_id = NEW.id; END;
